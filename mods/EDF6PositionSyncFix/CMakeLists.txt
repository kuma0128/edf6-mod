cmake_minimum_required(VERSION 3.24)
project(EDF6PositionSyncFix VERSION 0.2.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "EDF6 LocationSync requires Windows x64.")
endif()
add_library(edf6psf_core STATIC src/pattern_scan.cpp)
target_include_directories(edf6psf_core PUBLIC src)
target_compile_features(edf6psf_core PUBLIC cxx_std_17)
target_link_libraries(edf6psf_core PUBLIC bcrypt)
add_library(EDF6PositionSyncFix SHARED src/plugin.cpp src/exports.def)
target_link_libraries(EDF6PositionSyncFix PRIVATE edf6psf_core)
set_target_properties(EDF6PositionSyncFix PROPERTIES PREFIX "" OUTPUT_NAME "EDF6PositionSyncFix" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(all_targets edf6psf_core EDF6PositionSyncFix)
include(CTest)
if(BUILD_TESTING)
    add_library(fake_edf SHARED tests/fake_edf.cpp)
    set_target_properties(fake_edf PROPERTIES PREFIX "" OUTPUT_NAME "EDF" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/mock")
    list(APPEND all_targets fake_edf)
    add_executable(pattern_scan_tests tests/pattern_scan_tests.cpp)
    add_executable(player_hook_tests tests/player_hook_tests.cpp)
    add_executable(signature_check tools/signature_check.cpp)
    add_executable(plugin_load_smoke tests/plugin_load_smoke.cpp)
    foreach(t pattern_scan_tests player_hook_tests signature_check plugin_load_smoke)
        target_link_libraries(${t} PRIVATE edf6psf_core)
        list(APPEND all_targets ${t})
    endforeach()
    add_dependencies(plugin_load_smoke EDF6PositionSyncFix fake_edf)
    add_test(NAME pattern_scan COMMAND pattern_scan_tests)
    add_test(NAME player_hook COMMAND player_hook_tests)
    add_test(NAME plugin_load_smoke COMMAND plugin_load_smoke "$<TARGET_FILE:EDF6PositionSyncFix>" "$<TARGET_FILE:fake_edf>")
    if(MINGW)
        target_link_options(signature_check PRIVATE -municode)
        target_link_options(plugin_load_smoke PRIVATE -municode)
    endif()
endif()
foreach(t ${all_targets})
    target_compile_definitions(${t} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
    if(MSVC)
        target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8)
        set_property(TARGET ${t} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    elseif(MINGW)
        target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror -finput-charset=UTF-8 -fexec-charset=UTF-8)
        target_link_options(${t} PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
endforeach()
